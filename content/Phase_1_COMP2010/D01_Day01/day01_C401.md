### ## 🎯 1. Mục tiêu bài học & Thực hành (Codelab Objectives)

* **Tích hợp Multi-LLM API:** Làm việc với 3 nhà cung cấp hàng đầu: OpenAI, Google Gemini, và Anthropic Claude.
* **Xây dựng Streaming Chatbot:** Phát triển ứng dụng phản hồi thời gian thực qua terminal kết hợp quản lý lịch sử trò chuyện (Conversation History).
* **Token Economics & Chi phí:** Đo lường và tính toán chi phí gọi API thực tế trên mỗi triệu tokens (1M tokens) cho từng dòng mô hình.
* **Siêu tham số điều khiển (Hyperparameters):**
* `temperature`: Điều khiển độ ngẫu nhiên/sáng tạo (0.0 = xác định, 1.0+ = sáng tạo).
* `top_p` (Nucleus Sampling): Giới hạn tập token ứng viên dựa trên xác suất tích lũy.
* `max_tokens` / `max_output_tokens`: Giới hạn độ dài output phản hồi.
* `latency`: Thời gian phản hồi từ lúc gửi prompt đến khi nhận kết quả.

### Task 1: Tích hợp OpenAI API (GPT-4o / GPT-4o-mini)
```python
import os
import time
def call_openai(
prompt: str,
model: str = "gpt-4o",
temperature: float = 0.7,
top_p: float = 0.9,
max_tokens: int = 256,
) -> tuple[str, float, dict]:
"""
Gọi OpenAI API, đo độ trễ (latency) và thống kê token sử dụng.
"""
from openai import OpenAI
api_key = os.getenv("OPENAI_API_KEY") or "mock-key"
client = OpenAI(api_key=api_key)
start_time = time.time()
response = client.chat.completions.create(
model=model,
messages=[{"role": "user", "content": prompt}],
temperature=temperature,
top_p=top_p,
max_tokens=max_tokens,
)
latency = time.time() - start_time
text = response.choices[0].message.content or ""
usage = {
"input_tokens": response.usage.prompt_tokens if response.usage else 0,
"output_tokens": response.usage.completion_tokens if response.usage else 0,
}
return text, latency, usage
```

### ### Task 2: Tích hợp Google Gemini API (Dual SDK Fallback)

```python
import os
import time
def call_gemini(
prompt: str,
model: str = "gemini-2.5-flash",
temperature: float = 0.7,
top_p: float = 0.9,
max_tokens: int = 256,
) -> tuple[str, float, dict]:
"""
Gọi Google Gemini API hỗ trợ fallback giữa SDK mới (`google-genai`)
và SDK cũ (`google-generativeai`).
"""
api_key = os.getenv("GEMINI_API_KEY") or os.getenv("GOOGLE_API_KEY") or "mock-key"
start_time = time.time()
try:
# Chuẩn SDK mới: google-genai
from google import genai
from google.genai import types
client = genai.Client(api_key=api_key)
config = types.GenerateContentConfig(
temperature=temperature,
top_p=top_p,
max_output_tokens=max_tokens,
)
response = client.models.generate_content(
model=model,
contents=prompt,
config=config,
)
latency = time.time() - start_time
text = response.text or ""
usage = {
"input_tokens": response.usage_metadata.prompt_token_count if response.usage_metadata else 0,
"output_tokens": response.usage_metadata.candidates_token_count if response.usage_metadata else 0,
}
return text, latency, usage
except (ImportError, Exception):
# Fallback về legacy SDK: google-generativeai
import google.generativeai as genai
genai.configure(api_key=api_key)
model_inst = genai.GenerativeModel(model)
config = genai.types.GenerationConfig(
temperature=temperature,
top_p=top_p,
max_output_tokens=max_tokens,
)
response = model_inst.generate_content(prompt, generation_config=config)
latency = time.time() - start_time
text = response.text or ""
try:
input_tokens = model_inst.count_tokens(prompt).total_tokens
output_tokens = model_inst.count_tokens(text).total_tokens
except Exception:
input_tokens = int(len(prompt.split()) * 1.5)
output_tokens = int(len(text.split()) * 1.5)
usage = {
"input_tokens": input_tokens,
"output_tokens": output_tokens,
}
return text, latency, usage
```

### ### Task 4: So Sánh Đa Mô Hình & Tính Toán Chi Phí (Token Economics)

**Công thức tính chi phí:**
$$\\text{Total Cost} = \\frac{\\text{Input Tokens} \\times \\text{Price}_{\\text{Input, 1M}} + \\text{Output Tokens} \\times \\text{Price}_{\\text{Output, 1M}}}{1,000,000}$$
```python
PRICING_1M_TOKENS = {
"gpt-4o": {"input": 2.50, "output": 10.00},
"gpt-4o-mini": {"input": 0.15, "output": 0.60},
"gemini-2.5-flash": {"input": 0.075, "output": 0.30},
}
def compare_models(prompt: str) -> dict:
"""
Gọi song song GPT-4o, GPT-4o-mini và Gemini Flash cùng 1 prompt để đo lường
độ trễ (latency), chất lượng output và tổng chi phí tiêu thụ.
"""
# 1. GPT-4o
gpt4o_text, gpt4o_lat, gpt4o_usage = call_openai(prompt, model="gpt-4o")
gpt4o_cost = (
gpt4o_usage["input_tokens"] * PRICING_1M_TOKENS["gpt-4o"]["input"] +
gpt4o_usage["output_tokens"] * PRICING_1M_TOKENS["gpt-4o"]["output"]
) / 1_000_000
# 2. GPT-4o-mini
mini_text, mini_lat, mini_usage = call_openai(prompt, model="gpt-4o-mini")
mini_cost = (
mini_usage["input_tokens"] * PRICING_1M_TOKENS["gpt-4o-mini"]["input"] +
mini_usage["output_tokens"] * PRICING_1M_TOKENS["gpt-4o-mini"]["output"]
) / 1_000_000
# 3. Gemini 2.5 Flash
gemini_text, gemini_lat, gemini_usage = call_gemini(prompt, model="gemini-2.5-flash")
gemini_cost = (
gemini_usage["input_tokens"] * PRICING_1M_TOKENS["gemini-2.5-flash"]["input"] +
gemini_usage["output_tokens"] * PRICING_1M_TOKENS["gemini-2.5-flash"]["output"]
) / 1_000_000
return {
"gpt4o": {"response": gpt4o_text, "latency": gpt4o_lat, "cost": gpt4o_cost, "usage": gpt4o_usage},
"gpt4o_mini": {"response": mini_text, "latency": mini_lat, "cost": mini_cost, "usage": mini_usage},
"gemini_flash": {"response": gemini_text, "latency": gemini_lat, "cost": gemini_cost, "usage": gemini_usage},
}
```

### ### Task 5: Streaming Chatbot với Bộ Nhớ Ngữ Cảnh (Sliding Window Context)

```python
def streaming_chatbot() -> None:
"""
Chatbot dòng lệnh thời gian thực sử dụng Gemini 2.5 Flash Streaming.
Duy trì bộ nhớ trượt (Sliding Window) 3 lượt hội thoại gần nhất (6 tin nhắn).
"""
import os
from google import genai
from google.genai import types
api_key = os.getenv("GEMINI_API_KEY") or os.getenv("GOOGLE_API_KEY") or "mock-key"
client = genai.Client(api_key=api_key)
history = []  # Lưu trữ lịch sử: [{"role": "user"/"model", "text": "..."}]
print("🤖 Vin Smart Future Chatbot (Gemini 2.5 Flash Streaming)")
print("Gõ 'exit' hoặc 'quit' để kết thúc.\n")
while True:
try:
user_input = input("You: ").strip()
if not user_input:
continue
if user_input.lower() in ["quit", "exit"]:
break
# Lấy tối đa 3 lượt đối thoại gần nhất (6 messages)
messages_to_send = list(history[-6:])
messages_to_send.append({"role": "user", "text": user_input})
# Format sang Content Schema của Gemini SDK mới
formatted_contents = [
types.Content(
role=msg["role"],
parts=[types.Part.from_text(text=msg["text"])]
) for msg in messages_to_send
]
print("Gemini: ", end="", flush=True)
response_stream = client.models.generate_content_stream(
model="gemini-2.5-flash",
contents=formatted_contents
)
full_reply = ""
for chunk in response_stream:
chunk_text = chunk.text or ""
print(chunk_text, end="", flush=True)
full_reply += chunk_text
print("\n")
# Cập nhật lịch sử bộ nhớ
history.append({"role": "user", "text": user_input})
history.append({"role": "model", "text": full_reply})
except KeyboardInterrupt:
break
except Exception as e:
print(f"[Error]: {e}")
```

### A. Exponential Backoff Retry (Tự động thử lại khi API gặp lỗi Rate Limit)
```python
from typing import Callable, Any
import time
def retry_with_backoff(
fn: Callable[[], Any],
max_retries: int = 3,
base_delay: float = 0.1,
) -> Any:
"""
Gọi fn(). Tự động retry theo lũy thừa 2 (delay = base_delay * 2^attempt)
để xử lý lỗi nghẽn mạng hoặc Rate Limit (429).
"""
last_exception = None
for attempt in range(max_retries + 1):
try:
return fn()
except Exception as e:
last_exception = e
if attempt < max_retries:
delay = base_delay * (2 ** attempt)
time.sleep(delay)
if last_exception:
raise last_exception
```

```python
def format_comparison_table(results: list[dict]) -> str:
"""
Chuyển đổi danh sách kết quả đo kiểm thành bảng Markdown trực quan.
"""
table_lines = [
"| Prompt | Model | Response | Latency | Tokens (In/Out) | Cost (USD) |",
"| :--- | :--- | :--- | :--- | :--- | :--- |"
]
for r in results:
prompt = r["prompt"][:37] + "..." if len(r["prompt"]) > 40 else r["prompt"]
for name, data in [("GPT-4o", r.get("gpt4o")), ("GPT-4o-Mini", r.get("gpt4o_mini")), ("Gemini-Flash", r.get("gemini_flash"))]:
if not data:
continue
resp = data.get("response", "").replace("\n", " ")
resp_trunc = resp[:47] + "..." if len(resp) > 50 else resp
lat = data.get("latency", 0.0)
in_t = data.get("usage", {}).get("input_tokens", 0)
out_t = data.get("usage", {}).get("output_tokens", 0)
cost = data.get("cost", 0.0)
table_lines.append(f"| {prompt} | {name} | {resp_trunc} | {lat:.2f}s | {in_t}/{out_t} | ${cost:.6f} |")
return "\n".join(table_lines)
```
